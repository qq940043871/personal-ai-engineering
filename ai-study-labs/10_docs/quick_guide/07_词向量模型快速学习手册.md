# 词向量模型 快速学习手册

## 📋 概述

词向量模型（Embedding Models）将文本转换为高维向量表示，使计算机能够理解和处理语义信息，是 NLP 和 RAG 系统的核心组件。

## 🏗️ 核心概念

```
┌─────────────────────────────────────────────────────┐
│                   词向量工作原理                       │
├─────────────────────────────────────────────────────┤
│                                                     │
│  "国王" ──────────────────────────────→ [0.23, ...] │
│  "女王" ──────────────────────────────→ [0.25, ...] │
│                                                     │
│  向量空间距离: 语义相近的词距离更近                    │
│  king - queen + woman ≈ queen (类比推理)              │
│                                                     │
└─────────────────────────────────────────────────────┘
```

### 3. SiliconFlow Embeddings

```python
import requests
from typing import List

class SiliconFlowEmbeddings:
    """SiliconFlow 嵌入模型封装"""

    def __init__(self, api_key: str, model_id: str = "BAAI/bge-large-en-v1.5"):
        self.api_key = api_key
        self.model_id = model_id
        self.api_url = "https://api.siliconflow.cn/v1/embeddings"

    def embed_documents(self, texts: List[str]) -> List[List[float]]:
        """批量获取文档向量"""
        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {self.api_key}"
        }

        data = {
            "model": self.model_id,
            "input": texts
        }

        response = requests.post(self.api_url, headers=headers, json=data)
        response.raise_for_status()

        return [item["embedding"] for item in response.json()["data"]]

    def embed_query(self, text: str) -> List[float]:
        """获取单个查询向量"""
        return self.embed_documents([text])[0]

# 使用示例
embeddings = SiliconFlowEmbeddings(
    api_key="your-api-key",
    model_id="BAAI/bge-large-zh-v1.5"  # 中文模型
)

# 获取查询向量
query_vec = embeddings.embed_query("今天天气怎么样？")

# 批量获取文档向量
doc_vecs = embeddings.embed_documents([
    "今天天气很好",
    "明天可能会下雨",
    "今天是晴天"
])
```

### 4. 批量处理 + 相似度计算

```python
import numpy as np
from sentence_transformers import SentenceTransformer

model = SentenceTransformer('all-MiniLM-L6-v2')

def compute_similarity(text1, text2):
    """计算两个文本的相似度"""
    vec1 = model.encode([text1])
    vec2 = model.encode([text2])

    # 余弦相似度
    similarity = np.dot(vec1[0], vec2[0]) / (
        np.linalg.norm(vec1[0]) * np.linalg.norm(vec2[0])
    )
    return similarity

# 使用示例
texts = [
    "机器学习是人工智能的分支",
    "深度学习是机器学习的一部分",
    "今天天气很不错"
]

# 批量编码
embeddings = model.encode(texts)

# 两两计算相似度
for i in range(len(texts)):
    for j in range(i + 1, len(texts)):
        sim = compute_similarity(texts[i], texts[j])
        print(f"'{texts[i]}' vs '{texts[j]}': {sim:.4f}")
```

### 5. Ollama RAG 中的 Embedding 使用

```python
import chromadb
import requests
import uuid

def ollama_embedding(text):
    """获取文本向量"""
    res = requests.post(
        "http://localhost:11434/api/embeddings",
        json={"model": "nomic-embed-text", "prompt": text}
    )
    return res.json()["embedding"]

def init_vector_db(documents):
    """初始化向量数据库"""
    client = chromadb.PersistentClient(path='db/chroma_db')
    collection = client.get_or_create_collection(name='docs')

    ids = [str(uuid.uuid4()) for _ in documents]
    embeddings = [ollama_embedding(doc) for doc in documents]

    collection.add(
        ids=ids,
        documents=documents,
        embeddings=embeddings
    )

    return collection

def search_similar(query, collection, top_k=3):
    """相似性搜索"""
    query_vec = ollama_embedding(query)

    results = collection.query(
        query_embeddings=[query_vec],
        n_results=top_k
    )

    return results['documents'][0]

# 使用示例
docs = [
    "人工智能是指计算机系统模拟人类智能",
    "机器学习是实现人工智能的方法",
    "深度学习是机器学习的一个分支"
]

collection = init_vector_db(docs)

# 搜索
results = search_similar("什么是机器学习？", collection)
for i, doc in enumerate(results, 1):
    print(f"{i}. {doc}")
```

## 🔧 主流 Embedding 模型

### 1. 中文模型

| 模型 | 维度 | 特点 | 适用场景 |
|------|------|------|---------|

### Q: 相似度计算方式？

```python
import numpy as np

def cosine_similarity(a, b):
    """余弦相似度"""
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

def euclidean_distance(a, b):
    """欧氏距离"""
    return np.linalg.norm(np.array(a) - np.array(b))

def dot_product(a, b):
    """点积（未归一化）"""
    return np.dot(a, b)
```
