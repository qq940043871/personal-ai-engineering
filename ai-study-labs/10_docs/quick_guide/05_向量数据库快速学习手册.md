# 向量数据库 快速学习手册

## 📋 概述

向量数据库用于存储和检索高维向量 embedding，支持语义相似度搜索，是 RAG（检索增强生成）系统的核心组件。

## 🏗️ 核心概念

```
┌─────────────────────────────────────────────────────┐
│                 向量数据库工作原理                      │
├─────────────────────────────────────────────────────┤
│  原始文本 → Embedding模型 → 高维向量 → 存储到向量数据库 │
│                                                     │
│  查询文本 → Embedding模型 → 高维向量 → 相似度检索     │
│                                                     │
│  支持: 余弦相似度 │ 欧氏距离 │ 点积                   │
└─────────────────────────────────────────────────────┘
```

### 1. Chroma + Ollama 向量数据库

```python
import uuid
import chromadb
import requests

def ollama_embedding_by_api(text):
    """使用 Ollama API 获取文本向量"""
    res = requests.post(
        url="http://localhost:11434/api/embeddings",
        json={
            "model": "nomic-embed-text",
            "prompt": text
        }
    )
    return res.json()["embedding"]

# 初始化 Chroma 客户端
chient = chromadb.PersistentClient(path='db/chroma_db')

# 创建集合
collection = chient.get_or_create_collection(name='collection_v1')

# 准备数据
document_list = ["苹果", "西瓜", "水果"]
ids = [str(uuid.uuid4()) for _ in document_list]

# 生成向量
embeddings = [ollama_embedding_by_api(text) for text in document_list]

# 插入数据
collection.add(
    ids=ids,
    documents=document_list,
    embeddings=embeddings
)

# 查询
query_text = "苹果"
query_embedding = ollama_embedding_by_api(query_text)
results = collection.query(
    query_embeddings=[query_embedding],
    query_texts=[query_text],
    n_results=2
)

print(results)
```

### 2. LangChain + Chroma 集成

```python
from langchain_community.document_loaders import TextLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import HuggingFaceEmbeddings

# 1. 加载文档
loader = TextLoader("example.txt", encoding="utf-8")
documents = loader.load()

# 2. 分割文档
text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=1000,
    chunk_overlap=0
)
docs = text_splitter.split_documents(documents)

# 3. 创建 Embeddings
embeddings = HuggingFaceEmbeddings(model_name="damo/nlp_corom_sentence-embedding_chinese-base")

# 4. 创建向量数据库
db = Chroma.from_documents(docs, embeddings, persist_directory="./chroma_db")

# 5. 持久化
db.persist()

# 6. 查询示例
query = "什么是人工智能？"
docs = db.similarity_search(query, k=3)

for doc in docs:
    print(f"内容: {doc.page_content[:100]}...")
    print(f"来源: {doc.metadata}\n")
```

### 3. SiliconFlow Embeddings + Chroma

```python
from langchain_community.vectorstores import Chroma

class SiliconFlowEmbeddings:
    """SiliconFlow嵌入模型"""

    def __init__(self, api_key: str, model_id: str = "BAAI/bge-large-en-v1.5"):
        self.api_key = api_key
        self.model_id = model_id
        self.api_url = "https://api.siliconflow.cn/v1/embeddings"

    def embed_documents(self, texts: List[str]) -> List[List[float]]:
        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {self.api_key}"
        }

        data = {
            "model": self.model_id,
            "input": texts
        }

        response = requests.post(self.api_url, headers=headers, json=data)
        return [item["embedding"] for item in response.json()["data"]]

    def embed_query(self, text: str) -> List[float]:
        return self.embed_documents([text])[0]

# 使用
```

### 4. Ollama RAG 完整流程

```python
import requests
import chromadb
import uuid

def ollama_embedding_by_api(text):
    """获取文本向量"""
    res = requests.post(
        url="http://localhost:11434/api/embeddings",
        json={"model": "nomic-embed-text", "prompt": text}
    )
    return res.json()["embedding"]

def init_db():
    """初始化向量数据库"""
    chient = chromadb.PersistentClient(path='db/chroma_db')
    collection = chient.get_or_create_collection(name='collection_v1')

    # 文档数据
    document_list = file_chunk_list()  # 假设这个函数返回文档块列表
    ids = [str(uuid.uuid4()) for _ in document_list]
    embeddings = [ollama_embedding_by_api(text) for text in document_list]

    collection.add(ids=ids, documents=document_list, embeddings=embeddings)

def rag_run(query):
    """RAG 查询流程"""
    # 连接数据库
    chient = chromadb.PersistentClient(path='db/chroma_db')
    collection = chient.get_collection(name='collection_v1')

    # 查询向量
    query_embedding = ollama_embedding_by_api(query)
    results = collection.query(
        query_embeddings=[query_embedding],
        query_texts=[query],
        n_results=2
    )

    # 构建上下文
    context = "\n".join(results['documents'][0])

    # 构建 Prompt
    prompt_template = """你是专家，任务是根据参考信息回答用户问题。
    如果参考信息不足则回复"不知道"。
    请用中文回答。

    参考信息：
    {context}

    问题：{query}
    """

    prompt = prompt_template.format(context=context, query=query)
    return prompt
```

```python
# 基础搜索
results = collection.query(
    query_embeddings=[query_embedding],
    n_results=5
)

# 带过滤条件的搜索
results = collection.query(
    query_embeddings=[query_embedding],
    where={"metadata_field": "value"},  # 元数据过滤
    n_results=5
)

# MMR 搜索（最大边际相关）
results = collection.query(
    query_embeddings=[query_embedding],
    n_results=10,
    where_document={"$contains": "关键词"},  # 文档内容过滤
    n_results=5  # 最终返回数量
)
```

## 📊 主流向量数据库对比

| 数据库 | 特点 | 适用场景 | 部署难度 |
|--------|------|---------|---------|
| **Chroma** | 轻量级、本地存储 | 原型开发、小规模 | ⭐ 简单 |
| **Milvus** | 分布式、高可用 | 生产环境、大规模 | ⭐⭐⭐ 中等 |
| **Pinecone** | 云服务、完全托管 | 企业级应用 | ⭐ 简单 |
| **Weaviate** | 混合搜索、图结构 | 复杂检索场景 | ⭐⭐ 中等 |
| **Qdrant** | Rust 实现、高性能 | 实时检索 | ⭐⭐ 中等 |
