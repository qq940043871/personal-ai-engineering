### 2.2 工具选择算法——从 200+ 工具中找到正确的那个

**核心挑战**：工具定义长（平均 500 token/工具），全部放入上下文会爆 Token。

**解决方案：两阶段检索**。

#### 阶段1：离线索引构建

```python
# ========== 工具索引构建 ==========
from sentence_transformers import SentenceTransformer
from pymilvus import Collection, connections

# 1. 加载 Embedding 模型
embedder = SentenceTransformer('BAAI/bge-m3')

# 2. 构建工具元数据（用于检索）
tools = [
    {
        "name": "get_weather",
        "description": "获取指定城市的当前天气",
        "parameters": {
            "type": "object",
            "properties": {
                "city": {"type": "string", "description": "城市名称"}
            },
            "required": ["city"]
        },
        "tags": ["天气", "气象", "温度"]
    },
    # ... 199 more tools
]

# 3. 生成工具 Embedding（用于语义检索）
for tool in tools:
    # 融合描述 + 标签 + 参数描述
    doc = f"{tool['name']} {tool['description']} {' '.join(tool['tags'])}"
    embedding = embedder.encode(doc)
    # 存入 Milvus
    collection.insert([
        [tool['name']],  # 主键
        [embedding.tolist()],  # 向量
        [json.dumps(tool)]  # 原始JSON（用于返回）
    ])

# 4. 构建 BM25 索引（用于关键词检索）
from rank_bm25 import BM25Okapi
corpus = [f"{t['name']} {t['description']}" for t in tools]
tokenized_corpus = [doc.split() for doc in corpus]
bm25 = BM25Okapi(tokenized_corpus)
```

#### 阶段2：在线检索与排序

```python
# ========== 在线工具检索 ==========
def retrieve_tools(user_query: str, top_k: int = 10) -> List[Dict]:
    # 1. Dense 检索（语义相似度）
    query_embedding = embedder.encode(user_query)
    dense_results = collection.search(
        data=[query_embedding.tolist()],
        limit=100,
        output_fields=["tool_json"]
    )
    dense_tools = [res.entity.get('tool_json') for res in dense_results[0]]
    
    # 2. Sparse 检索（关键词匹配）
    tokenized_query = user_query.split()
    sparse_scores = bm25.get_scores(tokenized_query)
    sparse_top_100_idx = np.argsort(sparse_scores)[-100:][::-1]
    sparse_tools = [tools[i] for i in sparse_top_100_idx]
    
    # 3. 倒数排名融合（RRF）
    from collections import defaultdict
    rrf_scores = defaultdict(float)
    k = 60  # RRF 超参数
    
    # Dense 排名
    for rank, tool in enumerate(dense_tools):
        tool_id = tool['name']
        rrf_scores[tool_id] += 1 / (rank + k)
    
    # Sparse 排名
    for rank, tool in enumerate(sparse_tools):
        tool_id = tool['name']
        rrf_scores[tool_id] += 1 / (rank + k)
    
    # 排序
    merged_tools = sorted(rrf_scores.items(), key=lambda x: x[1], reverse=True)
    candidate_tools = [next(t for t in tools if t['name'] == tid) for tid, _ in merged_tools[:20]]
    
    # 4. Re-ranking（精排）
    from sentence_transformers import CrossEncoder
    reranker = CrossEncoder('BAAI/bge-reranker-base')
    
    # 构造 Query-Tool 对
    pairs = [[user_query, format_tool_for_reranker(t)] for t in candidate_tools]
    rerank_scores = reranker.predict(pairs)
    
    # 最终排序
    final_tools = [t for _, t in sorted(zip(rerank_score, candidate_tools), reverse=True)]
    return final_tools[:top_k]

def format_tool_for_reranker(tool: Dict) -> str:
    """格式化工具描述（用于Re-ranking）"""
    return f"工具名称：{tool['name']}\n描述：{tool['description']}\n参数：{json.dumps(tool['parameters'], ensure_ascii=False)}"
```

**性能优化**：

- Re-ranking 批量预测：一次推理处理 20 个候选（加速比 3x）

### 2.3 参数填充与校验——让工具调用真正可执行

**核心挑战**：用户表达模糊（"明天早上提醒我开会" → 需要解析时间 + 推断会议主题）

**解决方案：参数推断流水线**。

```python
# ========== 参数填充与校验系统 ==========
import json
from jsonschema import validate, ValidationError
from dateutil import parser as date_parser

class ParameterFiller:
    def __init__(self, tools: List[Dict]):
        self.tools = {t['name']: t for t in tools}
    
    def fill_parameters(self, tool_name: str, raw_params: Dict, context: Dict) -> Dict:
        """
        参数填充：结合原始参数 + 对话上下文
        """
        tool = self.tools[tool_name]
        schema = tool['parameters']
        properties = schema.get('properties', {})
        required = schema.get('required', [])
        
        filled_params = raw_params.copy()
        
        # 1. 必填参数检查与补全
        for param_name in required:
            if param_name not in filled_params:
                # 尝试从上下文推断
                inferred = self._infer_from_context(param_name, properties[param_name], context)
                if inferred is not None:
                    filled_params[param_name] = inferred
                else:
                    # 无法推断，返回错误
                    raise MissingParameterError(f"缺少必填参数：{param_name}")
        
        # 2. 参数类型转换
        for param_name, param_value in filled_params.items():
            if param_name in properties:
                expected_type = properties[param_name].get('type', 'string')
                filled_params[param_name] = self._convert_type(param_value, expected_type)
        
        # 3. JSON Schema 校验
        try:
            validate(instance=filled_params, schema=schema)
        except ValidationError as e:
            # 校验失败，让 LLM 重新生成（带错误反馈）
            raise SchemaValidationError(str(e))
        
        return filled_params
    
    def _infer_from_context(self, param_name: str, param_schema: Dict, context: Dict):
        """从对话上下文推断参数值"""
        # 示例：提醒场景
        if param_name == 'time':
            # 从上下文提取时间表达式
            time_expr = context.get('latest_time_expr')
            if time_expr:
                # 解析自然语言时间（"明天早上" → 2026-07-03 09:00）
                parsed_time = self._parse_time_expression(time_expr)
                return parsed_time.isoformat()
        
        if param_name == 'reminder_text':
            # 从上下文提取提醒内容
            return context.get('latest_user_intent', '')
        
        return None
    
    def _parse_time_expression(self, expr: str):
        """解析自然语言时间表达式"""
        # 使用 dateutil 解析
        # 增强：支持"明天"、"下周三"等中文表达
        import re
        if '明天' in expr:
            tomorrow = datetime.now() + timedelta(days=1)
            # 提取具体时间
            time_match = re.search(r'(\d{1,2})点', expr)
            if time_match:
                hour = int(time_match.group(1))
                return tomorrow.replace(hour=hour, minute=0, second=0)
            else:
                return tomorrow.replace(hour=9, minute=0, second=0)
        # ... 更多时间解析逻辑
    
    def _convert_type(self, value, expected_type: str):
        """参数类型转换"""
        if expected_type == 'integer':
            return int(value)
        elif expected_type == 'number':
            return float(value)
        elif expected_type == 'boolean':
            return bool(value)
        elif expected_type == 'array':
            if isinstance(value, str):
                return json.loads(value)
            return value
        else:  # string or object
            return value
```

### 2.4 多工具并行与依赖调度——DAG 动态调度器

**核心场景**：用户说"帮我查一下北京和上海的天气，然后告诉我哪个城市更适合明天出行"

**技术方案：DAG 调度器**。

```python
# ========== DAG 动态调度器 ==========
from typing import Dict, List, Any, Optional
from collections import defaultdict, deque
import json

class DAGScheduler:
    """
    基于 DAG 的工具调用调度器
    - 解析工具调用依赖
    - 并行执行无依赖的工具
    - 处理工具链（A的输出 → B的输入）
    """
    
    def __init__(self, tools: Dict[str, Dict]):
        self.tools = tools
    
    def build_dag(self, tool_calls: List[Dict]) -> Dict[str, List[str]]:
        """
        构建工具调用 DAG
        返回：{tool_call_id: [依赖的 tool_call_id 列表]}
        """
        dag = defaultdict(list)
        
        # 分析每个工具调用的输入依赖
        for i, call in enumerate(tool_calls):
            call_id = f"call_{i}"
            
            # 检查参数是否引用了前面工具的输出
            for param_name, param_value in call.get('parameters', {}).items():
                if isinstance(param_value, str) and param_value.startswith('${'):
                    # 引用格式：${call_0.result.field_name}
                    ref_call_id = param_value.split('.')[0].replace('${', '')
                    dag[call_id].append(ref_call_id)
        
        return dag
    
    def topological_sort(self, dag: Dict[str, List[str]]) -> List[str]:
        """拓扑排序（Kahn算法）"""
        in_degree = defaultdict(int)
        for node, deps in dag.items():
            in_degree[node]  # 确保节点存在
            for dep in deps:
                in_degree[node] += 1
        
        queue = deque([node for node in dag if in_degree[node] == 0])
        order = []
        
        while queue:
            node = queue.popleft()
            order.append(node)
            for neighbor in dag:
                if node in dag[neighbor]:
                    in_degree[neighbor] -= 1
                    if in_degree[neighbor] == 0:
                        queue.append(neighbor)
        
        return order
    
    def execute(self, tool_calls: List[Dict], llm, context: Dict) -> Dict[str, Any]:
        """
        执行工具调用 DAG
        """
        # 1. 构建 DAG
        dag = self.build_dag(tool_calls)
        
        # 2. 拓扑排序
        execution_order = self.topological_sort(dag)
        
        # 3. 按序执行（支持并行）
        results = {}
        
        # 分组：可以并行的放在同一层
        layers = self._group_into_layers(dag, execution_order)
        
        for layer in layers:
            # 同一层的工具调用可以并行
            import concurrent.futures
            with concurrent.futures.ThreadPoolExecutor() as executor:
                futures = {
                    executor.submit(
                        self._execute_single_tool,
                        call_id, tool_calls[int(call_id.split('_')[1])],
                        results, context
                    ): call_id for call_id in layer
                }
                
                for future in concurrent.futures.as_completed(futures):
                    call_id = futures[future]
                    results[call_id] = future.result()
        
        return results
    
    def _group_into_layers(self, dag: Dict[str, List[str]], order: List[str]) -> List[List[str]]:
        """将工具调用分组到不同层（同一层可并行）"""
        layer_for_node = {}
        for node in order:
            if not dag[node]:  # 无依赖
                layer_for_node[node] = 0
            else:
                max_dep_layer = max(layer_for_node[dep] for dep in dag[node])
                layer_for_node[node] = max_dep_layer + 1
        
        layers = defaultdict(list)
        for node, layer in layer_for_node.items():
            layers[layer].append(node)
        
        return [layers[i] for i in range(len(layers))]
    
    def _execute_single_tool(self, call_id: str, tool_call: Dict, previous_results: Dict, context: Dict):
        """执行单个工具调用"""
        tool_name = tool_call['tool']
        params = tool_call['parameters']
        
        # 解析参数中的引用（${call_0.result}）
        resolved_params = self._resolve_references(params, previous_results)
        
        # 执行
        result = execute_tool_with_recovery(tool_name, resolved_params, llm, context)
        return result
    
    def _resolve_references(self, params: Dict, results: Dict) -> Dict:
        """解析参数中的引用"""
        resolved = {}
        for key, value in params.items():
            if isinstance(value, str) and value.startswith('${'):
                # 解析引用
                ref_path = value[2:-1]  # 去掉 ${ 和 }
                parts = ref_path.split('.')
                result = results[parts[0]]  # call_0.result
                for part in parts[1:]:
                    if part == 'result':
                        continue
                    result = result[part]
                resolved[key] = result
            else:
                resolved[key] = value
        return resolved
```

## 📌 三、常见问答（Common Interview Q&A）

### Q1：Function Calling 和 ReAct 有什么区别？如何选择？

**五层深度答案**：

#### 第1层：概念区分

| 维度 | Function Calling | ReAct |
|------|------------------|-------|
| **定义** | LLM 输出结构化 JSON 指定工具名和参数 | LLM 交替生成推理轨迹（Thought）和动作（Action） |
| **输出格式** | 固定 JSON Schema | 自由文本（Thought: ... Action: ...） |
| **工具执行** | 由外部系统执行 | 由 Prompt 引导执行 |
| **可解析性** | 高（结构化输出） | 低（需要解析自由文本） |

#### 第2层：执行流程对比

**Function Calling 流程**：
```
用户请求 → LLM 生成 JSON → 外部系统解析 JSON → 执行工具 → 结果返回 LLM → 生成回复
```

**ReAct 流程**：
```
用户请求 → LLM 生成 Thought + Action → 解析 Action → 执行工具 → Observation → LLM 生成下一轮 Thought + Action → ... → 最终答案
```

#### 第3层：优缺点对比

**Function Calling 优势**：
1. 输出格式严格（100% 可解析）
2. 支持并行工具调用（DAG 调度）
3. 与 LLM API 深度集成（OpenAI Function Calling / vLLM JSON Mode）

**Function Calling 劣势**：
1. 需要微调或 Prompt Engineering 才能让模型学会格式
2. 不支持"思考过程"可视化（黑盒决策）

**ReAct 优势**：
1. 可解释性强（Thought 展示推理过程）
2. 灵活（可以自由设计 Action 格式）
3. 适合复杂多步推理（思维链 + 工具使用）

**ReAct 劣势**：
1. 输出解析困难（自由文本）
2. 多轮对话成本高（每轮都需要输入 Thought/Action 历史）

#### 第4层：选型决策树

```
是否需要并行工具调用？
  ├─ 是 → Function Calling
  └─ 否 → 继续
是否需要可视化推理过程？
  ├─ 是 → ReAct
  └─ 否 → Function Calling（推荐）
是否使用 OpenAI / vLLM API？
  ├─ 是 → Function Calling（原生支持）
  └─ 否 → ReAct（更灵活）
```

#### 第5层：混合方案（工业界最佳实践）

```python
# 混合方案：ReAct 用于规划 + Function Calling 用于执行
def hybrid_agent(user_query: str):
    # 第1阶段：ReAct 生成工具调用计划（DAG）
    react_prompt = f"""
    你是一个任务规划器。请将以下用户请求分解为工具调用序列。
    
    用户请求：{user_query}
    
    输出格式：
    Thought: <推理过程>
    Plan:
    1. <工具1> (<依赖>)
    2. <工具2> (<依赖>)
    ...
    """
    plan = llm.generate(react_prompt)
    
    # 第2阶段：解析 Plan → 工具调用 DAG
    dag = parse_plan_to_dag(plan)
    
    # 第3阶段：Function Calling 执行 DAG
    results = dag_scheduler.execute(dag, llm, context)
    
    # 第4阶段：汇总结果
    summary_prompt = f"""
    根据以下工具执行结果，生成用户友好的回复：
    {format_results(results)}
    """
    return llm.generate(summary_prompt)
```

**面试回答模板**：
"Function Calling 和 ReAct 不是非此即彼的关系，而是互补的。工业界最佳实践是用 ReAct 做规划（生成工具调用 DAG），用 Function Calling 做执行（精确调用工具）。这样既有可解释性（ReAct 的 Thought），又有执行可靠性（Function Calling 的结构化输出）。"

### Q2：如何实现工具调用的错误恢复？有哪些策略？

**四层深度答案**：

#### 第1层：错误分类

| 错误类型 | 占比 | 根因 | 恢复策略 |
|---------|------|------|---------|
| JSON 格式错误 | 15% | 模型未严格遵守格式 | JSON Force / Retry |
| 参数缺失 | 25% | 用户表达模糊 | 上下文推断 + 反问用户 |
| 参数类型错误 | 20% | 模型生成类型不匹配 | 类型转换 + Schema 校验 |
| 工具执行失败 | 30% | API 超时/授权失败 | 重试 + 降级策略 |
| 工具选择错误 | 10% | 检索召回错误 | 重新检索 + 用户确认 |

#### 第2层：错误恢复策略代码实现

```python
# ========== 多层错误恢复系统 ==========
class ErrorRecoverySystem:
    def __init__(self, llm, tool_registry):
        self.llm = llm
        self.tool_registry = tool_registry
        self.max_retry = 3
    
    def execute_with_recovery(self, tool_call: Dict, context: Dict) -> Dict:
        """带错误恢复的工具执行"""
        for attempt in range(self.max_retry):
            try:
                # 执行工具
                result = self.tool_registry.execute(
                    tool_call['tool'],
                    tool_call['parameters']
                )
                return result
            
            except json.JSONDecodeError as e:
                # 策略1：JSON 格式错误 → 让模型重新生成
                tool_call = self._recover_json_format(tool_call, str(e), context)
            
            except MissingParameterError as e:
                # 策略2：参数缺失 → 上下文推断 or 反问用户
                try:
                    tool_call['parameters'] = self._infer_missing_params(
                        tool_call['tool'],
                        tool_call['parameters'],
                        context
                    )
                except CannotInferError:
                    # 无法推断 → 反问用户
                    return self._ask_user_for_clarification(tool_call, str(e))
            
            except SchemaValidationError as e:
                # 策略3：Schema 校验失败 → 让模型修复参数
                tool_call = self._recover_schema_validation(tool_call, str(e), context)
            
            except ToolExecutionError as e:
                # 策略4：工具执行失败 → 重试 + 降级
                if attempt < self.max_retry - 1:
                    time.sleep(2 ** attempt)  # 指数退避
                    continue
                else:
                    # 最终降级策略
                    return self._fallback_strategy(tool_call, str(e), context)
        
        return {"error": "执行失败，超过最大重试次数"}
    
    def _recover_json_format(self, tool_call: Dict, error_msg: str, context: Dict) -> Dict:
        """恢复策略1：修复 JSON 格式"""
        prompt = f"""
        你生成的JSON格式有误：{error_msg}
        请重新生成严格的JSON格式（不要输出任何其他内容）。
        
        工具名称：{tool_call['tool']}
        用户请求：{context['user_query']}
        """
        new_response = self.llm.generate(prompt, json_mode=True)
        return json.loads(new_response)
    
    def _infer_missing_params(self, tool_name: str, params: Dict, context: Dict) -> Dict:
        """恢复策略2：推断缺失参数"""
        tool = self.tool_registry.get_tool(tool_name)
        required = tool['parameters'].get('required', [])
        
        for param in required:
            if param not in params:
                # 使用 LLM 从上下文推断
                infer_prompt = f"""
                用户请求：{context['user_query']}
                对话历史：{context['conversation_history']}
                
                请推断工具 "{tool_name}" 的必填参数 "{param}" 的值。
                只输出值，不要输出其他内容。
                """
                inferred_value = self.llm.generate(infer_prompt, max_tokens=50)
                params[param] = inferred_value.strip()
        
        return params
    
    def _ask_user_for_clarification(self, tool_call: Dict, error_msg: str) -> Dict:
        """恢复策略2b：反问用户"""
        missing_params = []  # 从 error_msg 解析
        question = f"我需要以下信息才能帮你：{', '.join(missing_params)}，请提供。"
        return {"need_user_input": True, "question": question}
    
    def _recover_schema_validation(self, tool_call: Dict, error_msg: str, context: Dict) -> Dict:
        """恢复策略3：修复 Schema 校验失败"""
        tool = self.tool_registry.get_tool(tool_call['tool'])
        prompt = f"""
        你生成的参数不符合工具Schema要求：{error_msg}
        
        工具定义：
        {json.dumps(tool, ensure_ascii=False)}
        
        请修复参数后重新生成完整的工具调用JSON。
        """
        new_response = self.llm.generate(prompt, json_mode=True)
        return json.loads(new_response)
    
    def _fallback_strategy(self, tool_call: Dict, error_msg: str, context: Dict) -> Dict:
        """恢复策略4：降级策略"""
        # 示例：天气工具失败 → 返回静态建议
        if tool_call['tool'] == 'get_weather':
            return {
                "fallback": True,
                "message": "抱歉，天气服务暂时不可用。建议你可以查看手机天气应用获取最新信息。"
            }
        
        # 通用降级：告诉用户失败原因
        return {
            "error": f"工具执行失败：{error_msg}",
            "suggestion": "请稍后重试，或联系技术支持。"
        }
```

#### 第3层：工程实践数据

| 策略 | 恢复成功率 | 平均延迟增加 |
|------|-----------|-------------|
| JSON Force（Grammar） | 99.5% | +50ms |
| 上下文推断 | 72% | +200ms |
| 重试（指数退避） | 85%（网络抖动场景） | +2s（重试等待） |
| 降级策略 | 100% | +0ms（返回静态响应） |

"工具调用的错误恢复是 Agent 系统可靠性的关键。我们采用了四层防御：

3. **人机协作层**：无法自动恢复时反问用户（用户体验保障）
4. **降级层**：工具失败时返回静态建议（系统可用性保障）

工程实践中，错误恢复的成功率从 60% 提升到了 92%。"

```python
def compute_parameter_accuracy(test_set: List[Dict]) -> float:
    """
    计算参数填充准确率（Exact Match + Partial Match）
    """
    exact_match = 0
    partial_match = 0
    total = len(test_set)
    
    for sample in test_set:
        predicted_params = sample['predicted_parameters']
        expected_params = sample['expected_parameters']
        
        # Exact Match：所有参数完全匹配
        if predicted_params == expected_params:
            exact_match += 1
            partial_match += 1
        else:
            # Partial Match：核心参数匹配（忽略可选参数）
            core_params = sample['tool_schema']['required']
            if all(predicted_params.get(p) == expected_params.get(p) for p in core_params):
                partial_match += 1
    
    return {
        'exact_match': exact_match / total,
        'partial_match': partial_match / total
    }
```

```python
def compute_e2e_success_rate(test_set: List[Dict]) -> float:
    """
    端到端评估：从用户请求到任务完成的成功率
    """
    success = 0
    total = len(test_set)
    
    for sample in test_set:
        user_query = sample['user_query']
        expected_outcome = sample['expected_outcome']
        
        # 执行完整流程
        try:
            actual_outcome = execute_user_query(user_query)
            
            # 判断任务是否成功（可能需要 LLM-as-Judge）
            if sample.get('need_judge', False):
                success_flag = llm_judge(expected_outcome, actual_outcome)
            else:
                success_flag = (actual_outcome == expected_outcome)
            
            if success_flag:
                success += 1
        except Exception as e:
            # 执行失败
            pass
    
    return success / total
```

###### 模块1：工具选择算法（支持 200+ 工具）

```python
class ToolRetriever:
    def __init__(self):
        # 离线索引
        self.embedder = SentenceTransformer('BAAI/bge-m3')
        self.milvus_collection = self._connect_milvus()
        self.bm25 = self._build_bm25_index()
        self.reranker = CrossEncoder('BAAI/bge-reranker-base')
    
    def retrieve(self, user_query: str, top_k: int = 10) -> List[Dict]:
        # 1. Dense 检索
        query_emb = self.embedder.encode(user_query)
        dense_results = self.milvus_collection.search(query_emb, limit=100)
        
        # 2. Sparse 检索
        sparse_scores = self.bm25.get_scores(user_query.split())
        sparse_top_100 = self._top_k_indices(sparse_scores, 100)
        
        # 3. RRF 融合
        rrf_scores = self._rrf_fusion(dense_results, sparse_top_100)
        
        # 4. Re-ranking
        candidates = self._get_top_n(rrf_scores, 20)
        rerank_scores = self.reranker.predict([
            [user_query, self._format_tool(t)] for t in candidates
        ])
        final_top_k = self._top_k_by_score(candidates, rerank_scores, top_k)
        
        return final_top_k
    
    def _rrf_fusion(self, dense_results, sparse_results, k: int = 60):
        """倒数排名融合"""
        scores = defaultdict(float)
        for rank, item in enumerate(dense_results):
            scores[item['id']] += 1 / (rank + k)
        for rank, idx in enumerate(sparse_results):
            item = self.tools[idx]
            scores[item['id']] += 1 / (rank + k)
        return scores
```

**性能优化**：
- Milvus 索引：IVF_FLAT（召回率 98%，查询延迟 8ms）
- Embedding 缓存：Redis（热点 Query 缓存，命中率 35%）
- Re-ranking 批量推理：一次处理 20 个候选（加速比 3x）

```python
class DAGScheduler:
    def execute(self, tool_calls: List[Dict]) -> Dict[str, Any]:
        # 1. 构建依赖图
        dag = self._build_dag(tool_calls)
        
        # 2. 分层（同一层可并行）
        layers = self._layering(dag)
        
        # 3. 逐层执行
        results = {}
        for layer in layers:
            # 并行执行当前层
            layer_results = self._parallel_execute(layer, results)
            results.update(layer_results)
        
        return results
    
    def _build_dag(self, tool_calls: List[Dict]) -> Dict[str, List[str]]:
        """构建 DAG（基于参数引用分析）"""
        dag = defaultdict(list)
        for i, call in enumerate(tool_calls):
            call_id = f"call_{i}"
            for param_value in call['parameters'].values():
                if isinstance(param_value, str) and param_value.startswith('${'):
                    dep_call_id = param_value.split('.')[0].replace('${', '')
                    dag[call_id].append(dep_call_id)
        return dag
```

class ErrorRecoverySystem:
    def execute_with_recovery(self, tool_call: Dict):
        for attempt in range(3):
            try:
                return execute_tool(tool_call)
            except JSONDecodeError:
                tool_call = self._fix_json(tool_call)
            except MissingParameterError:
                tool_call = self._infer_params(tool_call)
            except ToolExecutionError:
                if attempt < 2:
