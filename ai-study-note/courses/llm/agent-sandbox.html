1 沙箱安全全景图

Claude CLI 的 Agent 沙箱采用分层防御（Defense in Depth）设计。每个工具调用必须依次通过所有适用的安全层，任何一层拒绝即终止执行。

2 第一层：权限管控引擎

最外层的安全门。每次工具调用必须过权限检查，采用拒绝优先（Deny First）策略：一旦命中 Deny List，连弹窗都没有，直接驳回。

Allow — 自动放行

读操作（Read、Glob、Grep）默认允许。用户可通过 allowlist 预设更多自动放行规则。

Prompt — 弹窗确认

写操作（Edit、Write、Bash）默认弹窗。用户可选 Allow/Allow Always/Deny/Deny Always。

Deny — 拒绝执行

危险操作（如 rm -rf /*、curl | sh）被策略禁止。命中后直接拒绝，不弹窗。

// ============================================
// Layer 1: PermissionEngine — 权限管控引擎
// ============================================
 
class PermissionEngine {
 
constructor() {
// 权限策略：从 settings.json 加载
this.allowlist = [] // [{tool, pattern?}]
this.denylist = [] // [{tool, pattern?}]
this.sessionAllowCache = [] // 本次会话已确认的权限
this.permissionMode = 'normal' // permissive | normal | quiet
}
 
// ---------- 核心方法：检查权限 ----------
async checkPermission(toolName, args) {
 
// Step 1: 检查 Deny List（拒绝优先）
denyMatch = this.matchPolicy(this.denylist, toolName, args)
if (denyMatch) {
return {
decision: 'deny',
reason: `操作被策略禁止: ${denyMatch.pattern}`,
autoDeny: true // true = 不弹窗
}
}
 
// Step 2: 检查 Allow List
allowMatch = this.matchPolicy(this.allowlist, toolName, args)
if (allowMatch) {
return { decision: 'allow', autoAllow: true }
}
 
// Step 3: 检查本次会话是否已确认过
cached = this.sessionAllowCache.find(
c => c.tool === toolName && argsMatch(c.args, args)
)
if (cached) {
return { decision: 'allow', fromCache: true }
}
 
// Step 4: 根据 mode 决定是否弹窗
if (this.permissionMode === 'permissive') {
// 宽松模式：写操作也自动放行（不推荐生产）
return { decision: 'allow', mode: 'permissive' }
}
 
if (this.permissionMode === 'quiet') {
// 安静模式：直接拒绝未授权的所有操作
return { decision: 'deny', reason: '安静模式，未授权操作' }
}
 
// Step 5: 弹窗让用户选择
userChoice = await this.showPermissionDialog(toolName, args)
 
switch (userChoice.action) {
case 'allow':
return { decision: 'allow' }
case 'allow_always':
// 加入会话缓存
this.sessionAllowCache.push({ tool: toolName })
return { decision: 'allow', cached: true }
case 'deny':
return { decision: 'deny', reason: '用户拒绝' }
case 'deny_always':
this.denylist.push({ tool: toolName, pattern: toPattern(args) })
return { decision: 'deny', persistent: true }
case 'timeout':
// 用户超时未响应 → 默认拒绝
return { decision: 'deny', reason: '用户超时未响应' }
}
}
 
// ---------- 策略匹配 ----------
matchPolicy(policies, toolName, args) {
return policies.find(policy => {
// 工具名称必须匹配
if (policy.tool !== toolName) return false
// 如果有 pattern，对参数做模式匹配
if (policy.pattern) {
// 将参数序列化为字符串，正则匹配
argStr = JSON.stringify(args)
return new RegExp(policy.pattern).test(argStr)
}
return true // 无 pattern = 匹配该工具下所有调用
})
}
 
// ---------- 权限弹窗（UI 层） ----------
async showPermissionDialog(toolName, args) {
// 在终端中弹出一个交互式选择框
// 显示工具名称、参数预览、风险等级
return await terminal.showPrompt({
type: 'permission',
title: `允许执行 ${toolName}?`,
detail: formatArgs(args),
choices: [
{ label: '✅ 允许', value: 'allow' },
{ label: '✅ 始终允许', value: 'allow_always' },
{ label: '❌ 拒绝', value: 'deny' },
{ label: '🚫 始终拒绝', value: 'deny_always' },
],
timeout: 30_000 // 30s 超时
})
}
}

3 第二层：Bash 沙箱

Bash 是最危险的工具——它可以直接执行任意系统命令。因此它有最严格的多重保护。

⏱️ 超时控制

每个命令有默认超时（120s），防止死循环或挂起。通过 timeout 参数可自定义。超时后返回错误，不阻塞 Harness。

🔍 命令模式匹配

通过 pattern 字段匹配命令内容。粗粒度管控：rm -rf、curl | sh、dd 等危险命令默认被禁止。

🚫 沙箱模式开关

dangerouslyDisableSandbox: true 默认关闭，需要用户明确授权才能绕过。这是最后一道防线。

💻 非交互式 Shell

Bash 运行在非交互模式下（POSIX sh，不是 cmd.exe/PowerShell）。git rebase -i、vim 等交互式命令无法执行。

// ============================================
// Layer 2: BashSandbox — Shell 执行沙箱
// ============================================
 
class BashSandbox {
 
constructor(config) {
this.defaultTimeout = config.timeout || 120_000 // 默认 120s
this.sandboxEnabled = config.sandboxEnabled ?? true
this.dangerousPatterns = [
/rm\s+-rf\s+\/*/, // 删除根目录
/curl\s+.*\s*\|\s*(ba)?sh/, // curl 到 shell
/wget\s+.*\s*-O\s*-\s*\|\s*(ba)?sh/,
/dd\s+if=.*of=.*/, // 磁盘写入
/>\s*\/dev\//, // 直接写设备
]
}
 
async execute(command, options = {}) {
timeout = options.timeout || this.defaultTimeout
 
// Phase 1: 沙箱安全检查
if (this.sandboxEnabled) {
for (const pattern of this.dangerousPatterns) {
if (pattern.test(command)) {
throw new SandboxError(
`沙箱禁止的操作: ${pattern}`
)
}
}
}
 
// Phase 2: 检测是否是交互式命令
if (this.isInteractive(command)) {
return {
exitCode: 1,
stdout: '',
stderr: '交互式命令不被支持（无 TTY）'
}
}
 
// Phase 3: 执行（带超时）
try {
result = await execShell(command, {
timeout,
shell: this.getShellPath(), // Git Bash / sh
cwd: options.cwd
})
 
return {
exitCode: result.code,
stdout: result.stdout,
stderr: result.stderr,
duration: result.duration
}
 
} catch (err) {
if (err.name === 'TimeoutError') {
return {
exitCode: 124, // timeout exit code
stdout: '',
stderr: `命令超时 (${timeout}ms)`
}
}
throw err // 其他异常继续上抛
}
}
 
// ---------- 交互式命令检测 ----------
isInteractive(command) {
interactivePatterns = [
/git\s+rebase\s+-i/,
/git\s+commit\s+--amend/,
/vim?\s+/,
/nano\s+/,
/top\s*$/,
/less\s+/,
]
return interactivePatterns.some(p => p.test(command))
}
 
// ---------- 后台进程执行 ----------
async executeBackground(command, options = {}) {
// 后台进程在独立进程中运行，不阻塞主循环
pid = await spawnDetached(command, {
stdio: 'pipe',
cwd: options.cwd
})
return { pid, status: 'running' }
// 输出通过后续的 TaskOutput 工具获取
}
}

4 第三层：文件状态追踪

这是防止"脏写"的关键机制——强制要求 Read-before-Edit，确保每次写入都基于最新的文件状态。

Read("src/main.py") │ ▼ Harness 记录: [src/main.py] 已读，缓存内容摘要 │ ▼ Edit("src/main.py", "xxx" → "yyy") │ ▼ ┌── 检查: 是否先 Read 过该文件？ │ ├─ ✅ 是 → 允许写入，更新文件状态缓存 │ └─ ❌ 否 → 拒绝！返回错误："必须先 Read 该文件" │ ▼ Read("src/main.py") │ ▼ 直接从内存缓存返回（无需重新读磁盘） 保证读写一致性
// ============================================
// Layer 3: FileStateTracker — 文件状态追踪
// ============================================
 
class FileStateTracker {
 
constructor() {
this.fileStates = new Map() // path → FileState
this.auditLog = [] // 写操作审计日志
}
 
// ---------- 记录读操作 ----------
recordRead(filePath, content) {
existing = this.fileStates.get(filePath)
 
this.fileStates.set(filePath, {
path: filePath,
content,
contentHash: hashContent(content),
lastReadAt: now(),
version: (existing?.version || 0) + 1,
isDirty: false // 读操作不会标记为脏
})
}
 
// ---------- 写前检查 ----------
checkBeforeWrite(filePath, newContent) {
state = this.fileStates.get(filePath)
 
// 检查: 之前有没有 Read 过？
if (!state) {
throw new WriteWithoutReadError(
`必须先 Read 再编辑: ${filePath}`
)
}
 
// 检查: 文件在磁盘上是否已被外部修改？
diskContent = await readFileFromDisk(filePath)
diskHash = hashContent(diskContent)
 
if (diskHash !== state.contentHash) {
// 文件已被外部修改！有冲突风险
if (state.isDirty) {
// Agent 已修改过 + 外部也修改了 → 冲突
throw new FileConflictError(
`文件冲突: ${filePath} 已被外部修改`
)
}
// 只读未改 + 外部修改 → 重新 Read 更新缓存
this.recordRead(filePath, diskContent)
}
 
return { allowed: true, state }
}
 
// ---------- 记录写操作 ----------
recordWrite(filePath, result) {
state = this.fileStates.get(filePath)
if (state) {
state.content = result.content
state.contentHash = hashContent(result.content)
state.isDirty = true
state.lastWriteAt = now()
}
 
// 写入审计日志
this.auditLog.push({
timestamp: now(),
action: 'write',
filePath,
contentHash: hashContent(result.content)
})
}
 
// ---------- 从缓存获取文件内容 ----------
getCachedContent(filePath) {
state = this.fileStates.get(filePath)
if (state && state.isDirty) {
// 如果文件已被写入，距离上次写入不超过 5s，返回缓存
age = now() - state.lastWriteAt
if (age < 5000) {
return state.content // 缓存命中
}
}
return null // 缓存过期/无缓存
}
}

5 第四层：Worktree 隔离

最强大的隔离机制——将 Agent 限在独立的 Git Worktree 中运行，主分支完全不受影响。

主工作目录 (main) │ ├── .git/worktrees/wt_abc123/ ← 独立的 git 元数据 ├── wt_abc123/src/ ← 代码的隔离副本 │ └── Agent 在这个隔离目录中操作 ├ 可以任意修改文件 ├ 可以执行 git rebase、git reset ├ 可以试验破坏性操作 └ 主分支完全不受影响！ 如果 Agent 没有做有效修改 → worktree 自动清理 如果 Agent 做了有效修改 → 可选择性地合并到主分支
// ============================================
// Layer 4: WorktreeIsolation — Git 工作树隔离
// ============================================
 
class WorktreeIsolation {
 
constructor(repoPath) {
this.repoPath = repoPath
this.worktrees = new Map() // agentId → worktreePath
this.cleanupTimers = new Map()
}
 
// ---------- 创建隔离工作树 ----------
async createIsolatedWorktree(agentId, branch) {
// 生成唯一名称
worktreeName = `agent-${agentId}-${Date.now()}`
worktreePath = path.join(this.repoPath, '..', worktreeName)
 
// 创建新的 git worktree
await execGit([
'git', 'worktree', 'add',
'--detach', // 不切换分支
worktreePath,
branch || 'HEAD'
])
 
this.worktrees.set(agentId, {
path: worktreePath,
createdAt: now(),
hasChanges: false
})
 
// 设置自动清理（如果无修改）
this.scheduleCleanup(agentId, 3_600_000) // 1h 后清理
 
return {
worktreePath,
worktreeName
}
}
 
// ---------- 在隔离环境中执行 Agent ----------
async runInIsolation(agentId, agentFn) {
info = this.worktrees.get(agentId)
if (!info) throw new Error('Worktree not found')
 
// 保存当前目录，切换到 worktree
originalCwd = process.cwd()
process.chdir(info.path)
 
try {
// Agent 在隔离目录中执行
result = await agentFn()
 
// 检查是否有未提交的修改
status = await execGit(['git', 'status', '--porcelain'])
info.hasChanges = status.stdout.trim().length > 0
 
return {
result,
hasChanges: info.hasChanges,
worktreePath: info.path
}
} finally {
process.chdir(originalCwd)
}
}
 
// ---------- 清理 ----------
async cleanup(agentId) {
info = this.worktrees.get(agentId)
if (!info) return
 
if (info.hasChanges) {
// 有修改 → 保留，记录日志
logger.info(`Agent ${agentId} 有未提交修改，保留 worktree`)
return
}
 
// 无修改 → 安全删除
await execGit([
'git', 'worktree', 'remove',
'--force', info.path
])
this.worktrees.delete(agentId)
this.clearCleanupTimer(agentId)
}
}

6 第五层：MCP 进程隔离

MCP（Model Context Protocol）服务器运行在独立子进程中，通过 stdin/stdout JSON-RPC 通信，彼此之间没有共享内存。

┌──────────────────────────────────────────────┐ │ Claude CLI (主进程) │ │ │ │ ┌─────────────┐ ┌─────────────┐ ┌────────┐ │ │ │ Harness │ │ Loop │ │ Memory │ │ │ │ 执行框架 │ │ 主循环 │ │ 记忆 │ │ │ └──────┬──────┘ └─────────────┘ └────────┘ │ └─────────┼────────────────────────────────────┘ │ JSON-RPC over stdin/stdout │ ┌─────────┴────────────────────────────────────┐ │ MCP Server A (子进程) — 文件系统工具 │ │ ├ 只能操作授权目录 │ │ ├ 独立资源限制 │ │ └ 崩溃不影响主进程 │ ├──────────────────────────────────────────────┤ │ MCP Server B (子进程) — 数据库工具 │ │ ├ 只能连接配置的数据库 │ │ └ 完全隔离的进程空间 │ └──────────────────────────────────────────────┘
// ============================================
// Layer 5: MCPProcessManager — MCP 进程隔离
// ============================================
 
class MCPProcessManager {
 
constructor() {
this.servers = new Map() // serverId → MCPProcess
this.nextServerId = 0
}
 
// ---------- 启动 MCP 服务器（子进程） ----------
async startServer(name, command, args, env) {
serverId = `mcp-${name}-${++this.nextServerId}`
 
// 启动子进程
child = spawn(command, args, {
stdio: ['pipe', 'pipe', 'pipe'],
// 隔离的环境变量
env: {
...env,
MCP_SERVER_ID: serverId,
PATH: process.env.PATH, // 只继承必要变量
},
cwd: this.getServerWorkDir(name)
})
 
server = {
id: serverId,
name,
child,
stdin: child.stdin,
stdout: child.stdout,
tools: [], // 服务器注册的工具列表
status: 'starting',
lastHeartbeat: now()
}
 
// 监听子进程退出
child.on('exit', (code) => {
server.status = `exited:${code}`
logger.warn(`MCP ${name} 进程退出: ${code}`)
// 自动重启（可选）
})
 
// 等待工具列表注册
server.tools = await this.waitForToolRegistration(server)
server.status = 'running'
 
this.servers.set(serverId, server)
return serverId
}
 
// ---------- 通过 JSON-RPC 调用 MCP 工具 ----------
async callTool(serverId, toolName, args) {
server = this.servers.get(serverId)
if (!server || server.status !== 'running') {
return { error: 'MCP 服务器不可用' }
}
 
// 构建 JSON-RPC 请求
request = {
jsonrpc: '2.0',
id: generateId(),
method: 'tools/call',
params: { name: toolName, arguments: args }
}
 
// 通过 stdin 发送
server.stdin.write(JSON.stringify(request) + '\n')
 
// 通过 stdout 等待响应（带超时）
response = await readJSONRPCResponse(server.stdout, 30_000)
 
if (response.error) {
return { error: response.error.message }
}
 
server.lastHeartbeat = now()
return { result: response.result }
}
 
// ---------- 安全关闭 ----------
async shutdownServer(serverId) {
server = this.servers.get(serverId)
if (!server) return
 
// 发送关闭信号
server.child.kill('SIGTERM')
 
// 等待优雅退出
exited = await waitForExit(server.child, 5000)
if (!exited) {
server.child.kill('SIGKILL') // 强制结束
}
 
this.servers.delete(serverId)
}
}

7 第六层：上下文约束

在"应用逻辑"层的安全控制——防止 Agent 资源失控。这些不是"安全"的直译，但它们是生产级 Agent 防止失控的必需组件。

🔄 最大迭代次数

防止 Agent 陷入无限工具调用循环。单次交互默认最多 50 次工具调用，达到上限后强制结束，防止 Token 失控。

💰 Token 预算上限

用户可指定 +500k 等 Token 预算（仅 Claude Code 支持），超出后停止工作。Workflow 中也通过 budget.total/spent/remaining() 动态控制。

🧵 并发槽位限制

并发 Agent 调用上限为 min(16, CPU 核心数 - 2)，超出排队等待，防止资源耗尽导致系统响应变慢。

🛑 用户中断信号

用户可以随时通过快捷键或取消按钮中断正在执行的 Agent 循环。Harness 检测到中断信号后停止所有正在执行的工具调用。

// ============================================
// Layer 6: ContextGuardrails — 上下文约束
// ============================================
 
class ContextGuardrails {
 
constructor(config) {
this.maxIterations = config.maxIterations || 50
this.maxTokens = config.maxTokens || 200_000
this.maxConcurrency = config.maxConcurrency
|| Math.min(16, os.cpus().length - 2)
this.abortController = new AbortController()
this.tokenBudget = null // { total, spent, remaining }
}
 
// ---------- 检查是否应该继续循环 ----------
shouldContinue(iteration, tokensUsed) {
if (this.abortController.signal.aborted) {
return {
ok: false,
reason: '用户中断'
}
}
 
if (iteration >= this.maxIterations) {
return {
ok: false,
reason: `达到最大迭代次数 (${this.maxIterations})`
}
}
 
if (this.tokenBudget) {
remaining = this.tokenBudget.total - tokensUsed
if (remaining <= 0) {
return {
ok: false,
reason: 'Token 预算已耗尽'
}
}
}
 
return { ok: true }
}
 
// ---------- 并发槽位管理 ----------
acquireSlot() {
return new Promise(resolve => {
if (this.activeSlots < this.maxConcurrency) {
this.activeSlots++
resolve()
} else {
this.waitQueue.push(resolve) // 排队等待
}
})
}
 
releaseSlot() {
if (this.waitQueue.length > 0) {
next = this.waitQueue.shift()
next() // 唤醒下一个等待者
} else {
this.activeSlots--
}
}
}

8 完整安全链路伪代码

以下伪代码展示了整个 Harness 如何串联 6 层安全机制，执行一次完整的工具调用：

// ===============================================================
// SandboxedHarness — 6 层安全串联的完整工具执行
// ===============================================================
 
class SandboxedHarness {
 
constructor() {
this.permissionEngine = new PermissionEngine() // 第1层
this.bashSandbox = new BashSandbox() // 第2层
this.fileTracker = new FileStateTracker() // 第3层
this.worktreeIsolation = new WorktreeIsolation() // 第4层
this.mcpManager = new MCPProcessManager() // 第5层
this.guardrails = new ContextGuardrails() // 第6层
this.schemas = new Map() // Tool Schema 注册表
this.toolExecutors = new Map() // 工具执行器映射
}
 
// ---------- 注册工具 ----------
registerTool(name, schema, executor) {
this.schemas.set(name, schema)
// executor 有两种类型：本地执行 or MCP 代理
this.toolExecutors.set(name, executor)
}
 
// ---------- ⭐ 核心：安全地执行一次工具调用 ----------
async executeSafely(toolCall, context = {}) {
const {
iteration, // 当前循环次数
tokensUsed, // 已用 Token 数
agentId, // Agent ID（用于 Worktree）
isolationMode // 'worktree' | null
} = context
 
// ===== 第6层：上下文约束 =====
check = this.guardrails.shouldContinue(iteration, tokensUsed)
if (!check.ok) {
return { error: check.reason }
}
 
// ----- 解析工具调用 -----
toolName = toolCall.function.name
args = JSON.parse(toolCall.function.arguments)
 
// ----- 查找 Schema & 校验参数 -----
schema = this.schemas.get(toolName)
if (!schema) {
return { error: `未知工具: ${toolName}` }
}
valid = validateJSONSchema(schema.parameters, args)
if (!valid.ok) {
return { error: `参数校验失败: ${valid.error}` }
}
 
// ===== 第1层：权限管控 =====
permResult = await this.permissionEngine.checkPermission(toolName, args)
if (permResult.decision === 'deny') {
return {
error: `权限拒绝: ${permResult.reason}`,
permissionDenied: true
}
}
 
// ===== 第3层：文件状态追踪（写操作） =====
if (isWriteOperation(toolName)) {
try {
this.fileTracker.checkBeforeWrite(
args.file_path || args.path, args.content || args.new_string
)
} catch (err) {
return { error: err.message }
}
}
 
// ===== 第2层：Bash 沙箱 =====
if (toolName === 'Bash') {
return await this.executeBashSafely(args)
}
 
// ===== 第4层：Worktree 隔离 =====
executor = this.toolExecutors.get(toolName)
 
if (isolationMode === 'worktree' && agentId) {
return await this.worktreeIsolation.runInIsolation(
agentId,
() => this.runExecutor(toolName, executor, args)
)
}
 
// ----- 正常执行 -----
return await this.runExecutor(toolName, executor, args)
}
 
// ---------- 安全执行 Bash ----------
async executeBashSafely(args) {
command = args.command
options = { timeout: args.timeout, cwd: args.cwd }
 
// 检查沙箱模式
if (args.dangerouslyDisableSandbox) {
// 用户明确绕过沙箱 → 记录高危操作，仍然检查 denylist
logger.warn(`高危: 沙箱已禁用, 命令: ${command}`)
permResult = await this.permissionEngine.checkPermission('Bash', args)
if (permResult.decision === 'deny') {
return { error: `权限拒绝: ${permResult.reason}` }
}
return await execShell(command, options)
}
 
// 走标准沙箱路径
result = await this.bashSandbox.execute(command, options)
 
return {
exitCode: result.exitCode,
stdout: result.stdout,
stderr: result.stderr
}
}
 
// ---------- 执行器路由（本地/MCP） ----------
async runExecutor(toolName, executor, args) {
if (executor.type === 'mcp') {
// ===== 第5层：MCP 进程隔离 =====
result = await this.mcpManager.callTool(
executor.serverId, toolName, args
)
} else {
// 本地执行（内置工具）
result = await withTimeout(
executor.handler(args),
schema.timeout || 120_000
)
}
 
// 写操作后更新文件状态
if (isWriteOperation(toolName) && !result.error) {
this.fileTracker.recordWrite(
args.file_path || args.path, result
)
}
 
return {
role: 'tool',
tool_call_id: toolCall.id,
content: result.error
? `Error: ${result.error}`
: formatToolResult(result)
}
}
 
// ---------- 批量执行（支持并行） ----------
async executeBatch(toolCalls, context) {
// 区分读写：读操作可以并行，写操作串行
reads = toolCalls.filter(tc => isReadOperation(tc.function.name))
writes = toolCalls.filter(tc => !isReadOperation(tc.function.name))
 
readResults = await Promise.all(
reads.map(tc => this.executeSafely(tc, context))
)
 
writeResults = []
for (const tc of writes) {
writeResults.push(
await this.executeSafely(tc, context)
)
}
 
return [...readResults, ...writeResults]
}
}

9 与 Docker 沙箱对比

维度 Claude CLI 沙箱 Docker 容器
隔离粒度 进程级 + 文件级 + 权限级 操作系统级（完整内核命名空间）
资源限制 Token + 并发槽位 + 超时 CPU + 内存 + 网络 + 磁盘
安全模型 信任模型：用户可覆盖权限 强制隔离：默认拒绝所有
启动速度 毫秒级 秒级
主要目标 防止误操作和权限滥用 防止恶意代码逃逸
适用场景 开发辅助、代码分析 生产部署、不可信代码运行
权限层级 6 层嵌套：从弹窗到进程隔离 用户/命名空间/capabilities

🎯 核心设计理念

Claude CLI 的沙箱是面向信任增强的设计，不是面向完全隔离的设计。它假设用户和 Agent 之间有基本的信任基础，但通过 6 层机制防止"意外"和"越权"。而 Docker 是面向"零信任"的完全隔离。

一句话总结：Claude CLI 的沙箱不是阻止 Agent 做事，而是确保 Agent 做的每一件事都经过用户明确或隐式的授权，并且在可控范围内执行。

安全不是功能，而是架构 — 6 层防御，层层递进，守护 Agent 的每一次操作

权限管控 → Bash 沙箱 → 文件追踪 → Worktree 隔离 → MCP 进程隔 -> 上下文约束
